// 滚动吸附 / Scroll snap
// 素材包: lp (落地页与网页动效) · element 06 · 分类: 滚动效果
// 示例效果: 小幅滚动后页面自动补完剩余距离，下一节与视口顶部对齐。
// 效果规格:
//   - 各节与视口同高 168px，第一节浅蓝、第二节深蓝。
//   - 每 2.4 秒吸附一次，0.45–0.85 秒用户只滚 34px，三次方缓出。
//   - 0.9–1.5 秒页面自动移动剩余 134px，对齐下一节顶部，三次方缓出。
//   - 第二个 2.4 秒向上做相同动作回第一节，对应 scroll-snap-type: y mandatory。
// 适用场景: 适合像幻灯片一样逐屏阅读的页面。
// 标签: scroll-snap, snap, full-page, sections, css, mandatory, scroll
// 尺寸: 539x208 px · 循环周期: 4.8 s
// Needs: kit lp -> https://motion-prompt.knothhe.com/api/v1/kit/lp.js
// 动态示例: https://motion-prompt.knothhe.com/#078
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 06
// 06 スクロールスナップ（少しスクロールすると、次のセクションの頭までピタッと送られる）
CELLS[5] = (k, w, h, u) => {
  frame(k);
  const T = CYC / 2, lu = mod(u, T), dir = u < T ? 1 : -1, from = u < T ? 0 : VH, DRIFT = 34;
  const y = from + dir * (DRIFT * eOut(P(lu, 0.45, 0.85)) + (VH - DRIFT) * eOut(P(lu, 0.9, 1.5)));
  view(k, () => {
    k.save(); k.translate(0, -y);
    for (let i = 0; i < 2; i++) {
      const y0 = i * VH, dark = i === 1;
      k.fillStyle = dark ? C.ink : C.blueL; k.fillRect(0, y0, VW, VH);
      txt(k, '0' + (i + 1), 34, y0 + 70, `800 46px ${EN}`, C.acc);
      txt(k, dark ? '使ってみる' : '知ってもらう', 34, y0 + 102, `800 18px ${JP}`, dark ? '#fff' : C.ink);
      bars(k, 34, y0 + 118, 170, 2, dark ? 'rgba(255,255,255,.3)' : '#C3CDEA');
      pic(k, 262, y0 + 24, 220, 120, dark ? 2 : 0, 8);
    }
    k.restore();
    sbar(k, y / VH, 0.5);
  });
};
