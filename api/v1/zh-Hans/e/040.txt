// 定格 / Freeze frame
// 素材包: camera (镜头运动与转场) · element 16 · 分类: 拍摄风格与时间
// 示例效果: 人物跳到最高点时画面暂停，1.4 秒后从同一帧继续。
// 效果规格:
//   - 相机完全固定，高 1.3m，暂停的是素材时间。
//   - 1.6 秒暂停、3.0 秒恢复，共 1.4 秒；素材时间在 0–1.6 秒正常，定格时为 1.6 秒，之后为 u−1.4。
//   - 人物距 7m，素材时间 0.6–2.6 秒抛物线跳跃，最高 1.1m，恰在 50% 的最高点定格。
//   - 9 片落叶同时暂停，下落速度为慢动作示例的 1.5 倍。
//   - 定格时 5px 蓝色边框和底部暂停图标在 0.12 秒内淡入淡出，图标为 68×40px 白色圆角框内的两条蓝线。
//   - 右上角仅定格时显示 FREEZE，无交叉淡化复位。
// 适用场景: 适合强调决定性瞬间，或叠加姓名和字幕。
// 标签: freeze frame, still frame, pause, hold, stop motion, highlight
// 尺寸: 539x208 px · 循环周期: 9.4 s
// Needs: kit camera -> https://motion-prompt.knothhe.com/api/v1/kit/camera.js
// 动态示例: https://motion-prompt.knothhe.com/#040
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 16
// 16 フリーズフレーム
CELLS[15] = shot((k, w, h, u) => {
  // 跳躍の頂点で 1.4 秒止める
  const tj = u < 1.6 ? u : u < 3.0 ? 1.6 : u - 1.4;   // 実時間→映像時間
  const q = P(tj, 0.6, 2.6);
  const frozen = u >= 1.6 && u < 3.0;
  world(k, w, h, cam({ y: 1.3 }), { person: { z: 7, y: jumpY(q), armUp: q > 0 && q < 1 }, extra: leaves(tj, 1.5) });
  if (frozen) {
    const a = Math.min(1, (u - 1.6) / 0.12, (3.0 - u) / 0.12);
    k.save(); k.globalAlpha = a; k.strokeStyle = C.blue; k.lineWidth = 5; k.strokeRect(2.5, 2.5, w - 5, h - 5);
    k.fillStyle = 'rgba(255,255,255,.88)'; rr(k, w / 2 - 34, h - 58, 68, 40, 8); k.fill();
    k.fillStyle = C.blue; k.fillRect(w / 2 - 12, h - 49, 8, 22); k.fillRect(w / 2 + 4, h - 49, 8, 22); k.restore();
  }
  return frozen ? 'FREEZE' : '';
});
