// 手持镜头 / Handheld
// 素材包: camera (镜头运动与转场) · element 10 · 分类: 拍摄风格与时间
// 示例效果: 整体画面轻微不规则晃动，如同手持而非三脚架拍摄。
// 效果规格:
//   - 机位基本固定，偏航、俯仰、侧倾和高度分别叠加不同周期的正弦，T = 2π×u/4.7。
//   - 偏航（rad）= 0.016·sin(3T) + 0.008·sin(7T+1) + 0.004·sin(13T+2)，最大约 ±1.6°。
//   - 俯仰（rad）= 0.012·sin(4T+2) + 0.006·sin(9T+0.5) + 0.003·sin(17T+1.2)，最大约 ±1.2°。
//   - 侧倾（rad）= 0.02·sin(5T+1) + 0.008·sin(11T+3)，最大约 ±1.6°；高度 = 1.5 + 0.03·sin(6T)m。
//   - 人物静止于前方 6.5m，晃动持续，无缓动或停留。
//   - 频率倍数均为整数，每 4.7 秒无缝循环，右上角显示 HANDHELD。
// 适用场景: 适合纪录片感或现场感。
// 标签: handheld, camera shake, shaky, documentary, wobble, noise, realism
// 尺寸: 539x208 px · 循环周期: 9.4 s
// Needs: kit camera -> https://motion-prompt.knothhe.com/api/v1/kit/camera.js
// 动态示例: https://motion-prompt.knothhe.com/#034
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 10
// 10 手持ち
CELLS[9] = shot((k, w, h, u) => {
  const T = u * 2 * Math.PI / CYC;   // 周期をサイクルに合わせて継ぎ目なし
  const n = (a, b, c2) => Math.sin(T * a + b) * c2;
  const yaw = n(3, 0, .016) + n(7, 1, .008) + n(13, 2, .004), pitch = n(4, 2, .012) + n(9, .5, .006) + n(17, 1.2, .003), roll = n(5, 1, .02) + n(11, 3, .008);
  world(k, w, h, cam({ yaw, pitch, roll, y: 1.5 + n(6, 0, .03) }), { person: { z: 6.5 } });
  return 'HANDHELD';
});
