// 鱼眼镜头 / Fisheye lens
// 素材包: photo (摄影风格、光线与效果) · element 14 · 分类: 镜头与相机效果
// 示例效果: 街景中央鼓起，建筑竖线和道路弯曲，四角不动。
// 效果规格:
//   - 场景 2：街角，建筑、树和路灯（x = 114、292、494）位于 y = 132–162 的道路后；左前人物脚位 (176, 200)，高 124px。
//   - 距中心 d 的点移到原距离的 g 倍，g = (1 + k) / (1 + k·d²/R²)，R 为中心到角的距离 289px。
//   - k 从 0→1.25，中心放大 2.25 倍，角不移动。
//   - 轮廓每 6px 分段再变形，让直线变曲线。
//   - 右上焦距 24mm→8mm。
//   - 时间：0.4–1.4 秒效果渐入，保持至 4.0 秒，4.8 秒前消失；每轮 5 秒，三次方缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合滑板、音乐视频等大胆、趣味或窥视孔视角。
// 标签: fisheye, fisheye lens, barrel distortion, ultra wide, 8mm, distortion
// 尺寸: 539x208 px · 循环周期: 5 s
// Needs: kit photo -> https://motion-prompt.knothhe.com/api/v1/kit/photo.js
// 动态示例: https://motion-prompt.knothhe.com/#110
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 14
// 魚眼レンズ: 中心がふくらみ、建物の縦の線が外側へ丸く曲がる
CELLS[13] = cell((k, u) => {
  const a = env(u), kk = 1.25 * a, cx = CW / 2, cy = CH / 2, R2 = cx * cx + cy * cy;
  const M = kk < 0.002 ? null : (x, y) => { const dx = x - cx, dy = y - cy, g = (1 + kk) / (1 + kk * (dx * dx + dy * dy) / R2); return [cx + dx * g, cy + dy * g]; };
  stageStreet(k, BASE, { M });
  hud(k, `${Math.round(lerp(24, 8, a))}mm`);
});
