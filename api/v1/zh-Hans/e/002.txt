// 错峰出现 / Stagger
// 素材包: motion (动态图形动作) · element 02 · 分类: 入场与文字
// 示例效果: 图表中的柱形从左到右依次升起。
// 效果规格:
//   - 每 2.35 秒一轮。7 根柱形，宽 44、间距 20、高度为 52、84、68、112、92、136、158，蓝、珊瑚、黄、薄荷色循环。
//   - 第 i 根柱形（从 0 起）在 0.15 + 0.12·i 秒开始，0.5 秒内用 backOut（s = 2.6）从 0 长到完整高度，底部固定，超出后回稳。
//   - 白点半径 6.5，位于柱顶上方 16px，比柱形延后 0.34 秒，用弹簧 1 − e^(−7t)·cos(16t) 出现。
//   - 退场：柱形从 1.78 + 0.045·i 秒起用三次方缓入在 0.26 秒内缩回；白点从 1.70 + 0.045·i 秒起在 0.2 秒内先消失。
// 适用场景: 适合列表、柱状图和卡片依次入场，引导视线。
// 标签: stagger, sequence, delay, cascade, bar-chart, list, entrance
// 尺寸: 539x208 px · 循环周期: 4.7 s
// Needs: kit motion -> https://motion-prompt.knothhe.com/api/v1/kit/motion.js
// 动态示例: https://motion-prompt.knothhe.com/#002
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 02
// 02 スタガー（グラフの棒が、少しずつ時間をずらして順に立ち上がる）
CELLS[1] = (k, w, h, u) => {
  stage(k);
  const N = 7, BW = 44, G = 20, x0 = (w - (N * BW + (N - 1) * G)) / 2, base = h - 30;
  const HS = [52, 84, 68, 112, 92, 136, 158];
  k.fillStyle = C.navy3; rr(k, x0 - 18, base, N * BW + (N - 1) * G + 36, 5, 2.5); k.fill();
  const lu = mod(u, CYC / 2);
  for (let i = 0; i < N; i++) {
    const t0 = 0.15 + i * 0.12, g = backOut(P(lu, t0, t0 + 0.5), 2.6) * (1 - eIn(P(lu, 1.78 + i * 0.045, 2.04 + i * 0.045)));
    if (g <= 0.001) continue;
    const hh = HS[i] * g, x = x0 + i * (BW + G);
    k.fillStyle = PAL[i % 4]; rr(k, x, base - hh, BW, hh, [12, 12, 3, 3]); k.fill();
    // 棒の上の点（さらに遅れて出る）
    const d = spring(lu - t0 - 0.34, 16, 7) * (1 - eIn(P(lu, 1.7 + i * 0.045, 1.9 + i * 0.045)));
    if (d > 0.01) { k.fillStyle = '#fff'; k.beginPath(); k.arc(x + BW / 2, base - hh - 16, 6.5 * d, 0, 7); k.fill(); }
  }
};
