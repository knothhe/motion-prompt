// 色差 / Chromatic aberration
// 素材包: photo (摄影风格、光线与效果) · element 15 · 分类: 镜头与相机效果
// 示例效果: 街景分成红绿蓝以不同倍率重合，人物与建筑出现红青边，越靠两侧越大。
// 效果规格:
//   - 场景 2：街角，建筑、树和路灯（x = 114、292、494）位于 y = 132–162 的道路后；左前人物脚位 (176, 200)，高 124px。
//   - 图像分别与红 #FF0000、绿 #00FF00、蓝 #0000FF 相乘成三份，在黑底上相加。
//   - 以中心缩放：红 1 + 2s、绿 1 + s、蓝 1，s 从 0→0.02。
//   - 左右边缘红蓝相差约 11px，中心为 0。
//   - 时间：0.4–1.4 秒效果渐入，保持至 4.0 秒，4.8 秒前消失；每轮 5 秒，三次方缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合老镜头、VHS 或故障感的不完美质感。
// 标签: chromatic aberration, color fringing, rgb split, lens aberration, fringe, glitch
// 尺寸: 539x208 px · 循环周期: 5 s
// Needs: kit photo -> https://motion-prompt.knothhe.com/api/v1/kit/photo.js
// 动态示例: https://motion-prompt.knothhe.com/#111
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 15
// 色収差: 赤・緑・青を少しずつ違う大きさで重ね、画面の端ほど縁に赤と青のずれが出る
CELLS[14] = cell((k, u) => {
  const a = env(u), s = 0.02 * a, cx = CW / 2, cy = CH / 2;
  if (s < 0.0005) { stageStreet(k, BASE); return; }
  const [S, ks] = layer('ca'); stageStreet(ks, BASE);
  k.fillStyle = '#000000'; k.fillRect(0, 0, CW, CH);
  for (const [col, sc] of [['#FF0000', 1 + 2 * s], ['#00FF00', 1 + s], ['#0000FF', 1]]) {
    const [L, kl] = layer('ca-' + col); kl.drawImage(S, 0, 0); kl.globalCompositeOperation = 'multiply'; kl.fillStyle = col; kl.fillRect(0, 0, CW, CH); kl.globalCompositeOperation = 'source-over';
    k.save(); k.globalCompositeOperation = 'lighter'; k.translate(cx, cy); k.scale(sc, sc); k.translate(-cx, -cy); k.drawImage(L, 0, 0); k.restore();
  }
});
