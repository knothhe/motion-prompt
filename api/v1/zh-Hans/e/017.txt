// 遮罩转场 / Mask transition
// 素材包: motion (动态图形动作) · element 17 · 分类: 转场与强调
// 示例效果: 圆形或星形窗口扩大，露出其中的下一场景。
// 效果规格:
//   - 白天→夜晚→白天，每段 2.35 秒。每半段 0.45–1.6 秒转场，持续 1.15 秒，三次方缓入缓出。
//   - 下一场景叠在当前场景上，用形状裁剪；形状扩大直到覆盖整个画面。
//   - 白天→夜晚：以太阳中心（112, 104）为圆心，半径 0→470。
//   - 夜晚→白天：以月亮中心（427, 104）为中心的五角星，外半径 0→1150，同时旋转 1.2rad，约 69°。
//   - 扩大时形状边缘带宽 7 的白色描边。
// 适用场景: 适合以太阳、标志或按钮等画面内物体作为转场起点。
// 标签: mask, transition, iris, wipe, shape-wipe, reveal, scene-change
// 尺寸: 539x208 px · 循环周期: 4.7 s
// Needs: kit motion -> https://motion-prompt.knothhe.com/api/v1/kit/motion.js
// 动态示例: https://motion-prompt.knothhe.com/#017
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 17
// 17 マスクトランジション（丸や星の形の「窓」が広がり、その中に次の画面が見える）
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
