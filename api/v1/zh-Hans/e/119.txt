// 漏光 / Light leak
// 素材包: photo (摄影风格、光线与效果) · element 23 · 分类: 色调与后期风格
// 示例效果: 风景右缘出现缓慢摇摆的漏光带，内侧浅黄、边缘红色。
// 效果规格:
//   - 场景 3：左侧半身像头部中心 (140, 82)、半径 28px，右侧山、房屋、树和太阳，太阳中心 (432, 54)、半径 15px。
//   - 4 个竖长椭圆以右缘外 30px 为中心，每循环摆动 ±6px，垂直半径为画面高度 1.1 倍。
//   - 水平半径与颜色：250px #FFD089、35%；170px #FF9A3D、40%；95px #FF5A3A、45%；45px #FF3B30、50%，半径均从 0 增至目标。
//   - 时间：0.4–1.4 秒效果渐入，保持至 4.0 秒，4.8 秒前消失；每轮 5 秒，三次方缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合怀旧、夏日和偶然拍下的回忆感。
// 标签: light leak, film leak, vintage, analog, lomo, warm glow
// 尺寸: 539x208 px · 循环周期: 5 s
// Needs: kit photo -> https://motion-prompt.knothhe.com/api/v1/kit/photo.js
// 动态示例: https://motion-prompt.knothhe.com/#119
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 23
// ライトリーク: 右端から、フィルムに光がもれたような赤と橙のにじみが帯になって入る
CELLS[22] = cell((k, u) => {
  const a = env(u), w = Math.sin(u / CYC * TAU);
  stageLand(k, BASE);
  k.save();
  // 縦長の楕円を右端の外に中心を置いて重ねる（外ほど淡い黄、端ほど赤）
  const band = (rx, col, al) => { k.globalAlpha = al * a; k.fillStyle = col; k.beginPath(); k.ellipse(CW + 30 + 6 * w, CH / 2, rx * a, CH * 1.1, 0, 0, TAU); k.fill(); };
  band(250, '#FFD089', 0.35); band(170, '#FF9A3D', 0.4); band(95, '#FF5A3A', 0.45); band(45, '#FF3B30', 0.5);
  k.restore();
});
