// 擦除转场 / Wipe
// 素材包: camera (镜头运动与转场) · element 21 · 分类: 转场
// 示例效果: 垂直边缘从左向右移动，后方露出新画面。
// 效果规格:
//   - 1.7–2.8 秒转场，共 1.1 秒，边缘从左外侧 x = −4px 到右外侧 x = 543px。
//   - 三次方缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
//   - 边缘左侧显示新画面，右侧显示旧画面；两幅都不移动，通过裁剪揭示而非推动。
//   - 边缘为 4px 蓝色竖线，仅移动时可见。
//   - 两个画面的机位与主体均固定。
//   - 每 4.7 秒交替 A→B、B→A，往返 9.4 秒，右上角显示方向。
// 适用场景: 适合明确切换场景、话题或对比的两侧。
// 标签: wipe, linear wipe, reveal, clip, edge, left to right, transition
// 尺寸: 539x208 px · 循环周期: 9.4 s
// Needs: kit camera -> https://motion-prompt.knothhe.com/api/v1/kit/camera.js
// 动态示例: https://motion-prompt.knothhe.com/#045
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 21
// 21 ワイプ
CELLS[20] = shot(abCell((k, w, h, u, A, B) => {
  const p = eIO(P(u, 1.7, 2.8)), ex = p * (w + 8) - 4;
  A(k, w, h);
  if (p > 0) { k.save(); k.beginPath(); k.rect(0, 0, ex, h); k.clip(); k.fillStyle = '#fff'; k.fillRect(0, 0, w, h); B(k, w, h); k.restore(); }
  if (p > 0 && p < 1) { k.fillStyle = C.blue; k.fillRect(ex - 2, 0, 4, h); }
}));
