// 运动拖尾 / Motion trail
// 素材包: motion (动态图形动作) · element 14 · 分类: 运动质感
// 示例效果: 圆球沿 8 字路径运动，身后留下变淡的副本。
// 效果规格:
//   - 珊瑚色球半径 25，白色中心半径 11，每 4.7 秒绕 8 字一周：x = 196·sin(a)，y = 62·sin(2a)，a 匀速增长。
//   - 15 个副本，每个沿轨迹落后 0.105rad，约 0.079 秒。
//   - 副本半径 22→6，透明度 0.9 × (1 − i/15) + 0.05，越旧越小越淡。
//   - 从最新到最旧，副本颜色珊瑚→黄→蓝。
//   - 先画最旧副本，最后画球体。
// 适用场景: 适合飞行物、光点或光标的速度感与轨迹。
// 标签: trail, motion-trail, echo, afterimage, ghost, path, speed
// 尺寸: 539x208 px · 循环周期: 4.7 s
// Needs: kit motion -> https://motion-prompt.knothhe.com/api/v1/kit/motion.js
// 动态示例: https://motion-prompt.knothhe.com/#014
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 14
// 14 モーショントレイル（動いた跡に、残像が尾を引く）
CELLS[13] = (k, w, h, u) => {
  stage(k);
  const X = w / 2, Y = h / 2 + 2, A = 196, B = 62, th = u / CYC * Math.PI * 2;
  const pt = a => [X + A * Math.sin(a), Y + B * Math.sin(2 * a)];
  const N = 15;
  for (let i = N; i >= 1; i--) {
    const [x, y] = pt(th - i * 0.105), f = i / N;
    k.globalAlpha = 0.9 * (1 - f) + 0.05; k.fillStyle = f < 0.5 ? mix(C.coral, C.yel, f * 2) : mix(C.yel, C.blue, (f - 0.5) * 2);
    k.beginPath(); k.arc(x, y, lerp(22, 6, f), 0, 7); k.fill();
  }
  k.globalAlpha = 1; const [x, y] = pt(th);
  k.fillStyle = C.coral; k.beginPath(); k.arc(x, y, 25, 0, 7); k.fill();
  k.fillStyle = '#fff'; k.beginPath(); k.arc(x, y, 11, 0, 7); k.fill();
};
