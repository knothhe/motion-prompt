// 遮罩揭示 / Mask reveal
// 素材包: motion (动态图形动作) · element 05 · 分类: 入场与文字
// 示例效果: 标题文字从一条分隔线后方滑出。
// 效果规格:
//   - 每 2.35 秒一轮。珊瑚色分隔线 380×6，y = 124，在 0.1–0.5 秒从中心展开，三次方缓入缓出；2.05–2.3 秒收回。
//   - 上方文字白色粗体 98px，裁剪范围 y = 8–119，在 0.35–1.0 秒从下方 120px 移到原位，五次方缓出。
//   - 下方文字黄色 32px，裁剪范围 y = 129–181，在 0.55–1.15 秒从上方 60px 向下归位，五次方缓出。
//   - 退场：上方文字在 1.7–2.05 秒向上 150px，backIn（s = 1.2），先轻微反向；下方文字在 1.75–2.05 秒向下 70px，三次方缓入。
//   - 不改变透明度，只通过离开裁剪区隐藏文字。
// 适用场景: 适合干净、稳重的标题入场。
// 标签: mask, reveal, clip, wipe, title, text, slide-up, headline
// 尺寸: 539x208 px · 循环周期: 4.7 s
// Needs: kit motion -> https://motion-prompt.knothhe.com/api/v1/kit/motion.js
// 动态示例: https://motion-prompt.knothhe.com/#005
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 05
// 05 マスクリビール（見えない枠＝線の向こうから、見出しがせり上がる）
CELLS[4] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 2, lu = mod(u, T), X = w / 2, LY = 124;
  const lg = eIO(P(lu, 0.1, 0.5)) * (1 - eIO(P(lu, 2.05, 2.3)));
  // 上の段：線の下から上へ
  const up = (1 - eOut5(P(lu, 0.35, 1.0))) * 120 + backIn(P(lu, 1.7, 2.05), 1.2) * -150;
  k.save(); k.beginPath(); k.rect(0, 8, w, LY - 8 - 5); k.clip();
  txt(k, 'Hello!', X, LY - 14 + up, `800 98px ${DISP}`, '#fff', 'center', 'alphabetic', '1px');
  k.restore();
  // 下の段：線の上から下へ
  const dn = (1 - eOut5(P(lu, 0.55, 1.15))) * -60 + eIn(P(lu, 1.75, 2.05)) * 70;
  k.save(); k.beginPath(); k.rect(0, LY + 5, w, 52); k.clip();
  txt(k, 'はじめまして', X, LY + 44 + dn, `700 32px ${JP}`, C.yel, 'center', 'alphabetic', '6px');
  k.restore();
  if (lg > 0.005) { k.fillStyle = C.coral; rr(k, X - 190 * lg, LY - 3, 380 * lg, 6, 3); k.fill(); }
};
