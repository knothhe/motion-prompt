// 悬停图片放大 / Image zoom on hover
// 素材包: lp (落地页与网页动效) · element 10 · 分类: 悬停与点击交互
// 示例效果: 光标下照片缓慢放大，边框不变，移开后缩回。
// 效果规格:
//   - 两张图片框 222×118、圆角 8，光标依次在左右各停约 1.5 秒。
//   - 内容以中心为锚点，在 0.5 秒三次方缓出缩放 1→1.15，移开后 0.5 秒恢复。
//   - 框外内容裁剪，相当于 overflow: hidden，框与说明文字不动。
// 适用场景: 适合作品、产品或文章图片网格，提示即将选择的项目。
// 标签: hover, zoom, image, scale, overflow-hidden, thumbnail, gallery
// 尺寸: 539x208 px · 循环周期: 4.8 s
// Needs: kit lp -> https://motion-prompt.knothhe.com/api/v1/kit/lp.js
// 动态示例: https://motion-prompt.knothhe.com/#082
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面仅由 u 决定。
// 以下为效果代码。CELLS、常量与辅助函数位于 kit 中。

// @element 10
// 10 ホバーで画像ズーム（枠はそのままで、中の写真だけがゆっくり拡大する）
CELLS[9] = (k, w, h, u) => {
  frame(k);
  const FX = [22, 267], FY = 16, FW = 222, FH = 118, ZOOM = 0.15;
  const cur = t => path([[0, 60, 160], [0.5, 120, 70], [1.8, 140, 84], [2.2, 370, 70], [3.4, 390, 84], [3.9, 470, 160], [4.8, 60, 160]], t);
  view(k, () => {
    for (let i = 0; i < 2; i++) {
      const inside = t => { const [x, y] = cur(t); return x > FX[i] && x < FX[i] + FW && y > FY && y < FY + FH; };
      const q = trans(inside, u, 0.5);
      pic(k, FX[i], FY, FW, FH, i, 8, 1 + ZOOM * q);
      txt(k, ['夏の山小屋', '海辺のカフェ'][i], FX[i], FY + FH + 20, `800 13px ${JP}`, C.ink); bars(k, FX[i], FY + FH + 28, 150, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
